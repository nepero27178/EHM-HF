\documentclass[tikz,border=3.14mm]{standalone}
\usepackage{tikz-3dplot}
\usetikzlibrary{backgrounds}

\definecolor{tabred}{RGB}{214, 39, 40}
\definecolor{tabblue}{HTML}{1f77b4}
\definecolor{tabgreen}{HTML}{2ca02c}

\begin{document}
\tdplotsetmaincoords{75}{110}
\begin{tikzpicture}[tdplot_main_coords,scale=1.2]

% Define layer spacing
\def\layersep{1}
\def\bondlength{1} % Length of a bond
\def\extension{0.3*\bondlength}

% Draw layers
\foreach \z in {0,2,4} {
    % Solid plane (opacity) at z-1, only for z=2,4
    \ifnum \z>0
        \filldraw[fill=lightgray, opacity=0.5, draw=none]
            (-2.5-\extension,-2.5-\extension,\z*\layersep-1) -- (2.5+\extension,-2.5-\extension,\z*\layersep-1) -- (2.5+\extension,2.5+\extension,\z*\layersep-1) -- (-2.5-\extension,2.5+\extension,\z*\layersep-1) -- cycle;
    \fi

    % Gray bonds (5x5 grid with extensions)
    \foreach \x in {-2,-1,0,1} {
        \foreach \y in {-2,-1,0,1,2} {
            \draw[lightgray] (\x-\extension,\y,\z*\layersep) -- (\x+1+\extension,\y,\z*\layersep);
        }
    }
    \foreach \y in {-2,-1,0,1} {
        \foreach \x in {-2,-1,0,1,2} {
            \draw[lightgray] (\x,\y-\extension,\z*\layersep) -- (\x,\y+1+\extension,\z*\layersep);
        }
    }

    % Copper oxide layer (5x5)
    \foreach \x in {-2,-1,0,1,2} {
        \foreach \y in {-2,-1,0,1,2} {
            % Copper atoms (red circles)
            \filldraw[tabred] (\x,\y,\z*\layersep) circle (0.1);
            % Oxygen atoms (blue circles) - between coppers
            \ifnum \x<2
                \filldraw[tabblue] (\x+0.5,\y,\z*\layersep) circle (0.06);
            \fi
            \ifnum \y<2
                \filldraw[tabblue] (\x,\y+0.5,\z*\layersep) circle (0.06);
            \fi
        }
    }
}

\end{tikzpicture}
\end{document}
