\begin{frame}
	\frametitle{The phase diagram of cuprates}
	
	Cuprates are the prototypical example of \textbf{strongly correlated materials}:\bigskip
	
	\begin{columns}
		\column{0.45\linewidth}
		\begin{itemize}
			\item \textbf{Antiferromagnetism} at null and low doping;
			\item \textbf{$d$-wave superconducting} domes.
		\end{itemize}
		Both arise from \textbf{strong electronic correlations}.\smallskip
		
		\DownImplication\smallskip
		
		Failure of band theory in microscopic description.
		\column{0.45\linewidth}
		\begin{figure}
			\includegraphics[width=\columnwidth]{\SectionFolder/phase-diagram.pdf}
			\caption*{Schematic phase diagram of cuprates. Author: Holger Motzkau, \hyperlink{https://commons.wikimedia.org/w/index.php?curid=27862441}{Wikimedia commons}, distributed according to \href{https://creativecommons.org/licenses/by-sa/4.0/deed.en}{the license CC BY-SA 4.0}}
		\end{figure}
	\end{columns}
\end{frame}

		\begin{itemize}
			\item $2$D copper oxide active layers;
			\item intermediate layers of inert material.
		\end{itemize}\medskip
		
		\DownImplication
		\column{0.48\linewidth}
		\begin{figure}
			\centering
			\def\GraphicsFolder{../Thesis/chapters/theoretical-introduction/pictures}
			\resizebox{\columnwidth}{!}{
				\input{\GraphicsFolder/copper-oxides}
			}
		\end{figure}
	\end{columns}
	\bigskip
	
	\begin{tabbox}[frametitle=The single-band Hubbard model on the square lattice]
		\[
		\hat H_\mathrm{HM} = 
		\underbrace{
			-t \sum_{\langle ij \rangle} \sum_\sigma \hat c_{i\sigma}^\dagger \hat c_{j\sigma}
		}_\text{Kinetic operator}
		\underbrace{
			+
			\vphantom{\sum_{\langle ij \rangle}}
			U \sum_i \hat n_{i\uparrow} \hat n_{i\downarrow}
		}_\text{On-site repulsion}
		\qquad
		\text{with $t, U  > 0$.}
		\]
	\end{tabbox}
\end{frame}

\begin{frame}
	\frametitle{Hubbard physics and cuprates}
	
	Despite its simplicity, the Hubbard model describes:
	\begin{itemize}
		\item Insulation at half-filling (\textbf{Mott localisation});
		\item \textbf{Antiferromagnetism} induced by Mott localisation;
		\item \textbf{$d$-wave superconductivity} emerging by doping the antiferromagnetic phase.
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Recent developments: the extended Hubbard model}
	\begin{columns}
		\column{0.3\linewidth}
		\begin{figure}
			\centering
			\includegraphics[width=0.7\columnwidth]{\SectionFolder/rixs-dmrg.png}
			\caption*{Hari Padma et al., 2025. \textbf{Physical Review X 15 (2): 021049}.
			}
		\end{figure}
		\column{0.6\linewidth} 
		\textbf{Observed}: incompatibilities of the Hubbard model with the spectrum of magnetic excitation in cuprate $\mathrm{Sr}_{14}\mathrm{Cu}_{24}\mathrm{O}_{41}$.
		
		\uncover<2>{
			\DownImplication
			\textbf{Proposal}: extension of the model with a \textbf{non-local attraction},
			\[
				\hat H_\mathrm{EHM} = \hat H_\mathrm{HM} \underbrace{
					-V \sum_{\langle ij \rangle} \hat n_i \hat n_j
				}_{\hat H_V} \, .
			\]
			$\hat H_V$ can act \textbf{as an independent source of $d$-wave superconductivity}.
		}
	\end{columns}
	
\end{frame}

\begin{frame}
	\frametitle{This thesis: investigating Mott physics in the EHM}
	\textbf{The model (EHM)}:
	\begin{equation*}
		\hat H_\mathrm{EHM} = 
		\underbrace{
			-t \sum_{\langle ij \rangle} \sum_\sigma \hat c_{i\sigma}^\dagger \hat c_{j\sigma}
		}_\text{Kinetic operator}
		\underbrace{
			+
			\vphantom{\sum_{\langle ij \rangle}}
			U \sum_i \hat n_{i\uparrow} \hat n_{i\downarrow}
		}_\text{Local repulsion} 
		\underbrace{
			-V \sum_{\langle ij \rangle} \hat n_i \hat n_j
		}_\text{Non-local attraction}
		\qquad
		\text{with $t, U, V  > 0$.}
	\end{equation*}

	\vspace{1em}

	\textbf{The motivation}: compatibility with some aspects of cuprates physics, including $d$-wave superconductivity.
	
	\vspace{1em}
	
	\textbf{The target}: understand how the new interaction relates to Mott physics, antiferromagnetism and superconductivity.
	
	\vspace{1em}
	
	\textbf{The method}: Hartree-Fock variational method, at low temperature.
\end{frame}
