\section{Backup 2: details on the HF method}

\begin{frame}
	\frametitle{The Hartree-Fock method in a nutshell}
	
	\textbf{Hartree-Fock method}: $\mathcal{S}$ is the set of all \textbf{gaussian states}, i.e. states that are non-interacting for some fermionic operators 
	\[
	\ket{\Psi} = \bigotimes_\alpha \hat \gamma_\alpha^\dagger \ket{\Omega_\gamma} \, ,
	\qquad
	\acomm{\hat \gamma_\alpha}{\hat \gamma_\beta^\dagger} = \delta_{\alpha\beta} \, ,
	\]
	with $\hat \gamma \leftrightarrow \hat c$ through a unitary map.\medskip
	\DownImplication
	\textbf{Wick's theorem}: averages of quartic operators decompose nicely,
	\[
	\langle
	\hat c_\alpha^\dagger \hat c_\beta^\dagger \hat c_\gamma \hat c_\delta
	\rangle = \underbrace{
		\langle 
		\hat c_\alpha^\dagger \hat c_\delta
		\rangle \langle	
		\hat c_\beta^\dagger \hat c_\gamma 
		\rangle
	}_{\text{Hartree}}
	- 
	\underbrace{
		\langle 
		\hat c_\alpha^\dagger \hat c_\gamma
		\rangle \langle	
		\hat c_\beta^\dagger \hat c_\delta 
		\rangle 
	}_{\text{Fock}}
	+ 
	\underbrace{
		\langle 
		\hat c_\alpha^\dagger \hat c_\beta^\dagger
		\rangle \langle	
		\hat c_\gamma \hat c_\delta 
		\rangle 
	}_{\text{Cooper}}.
	\]
	The EHM hamiltonian is a quartic operator!
\end{frame}

\begin{frame}
	\frametitle{The self-consistency mapping}
	Each $\ket{\Psi}\in\mathcal{S}$ is uniquely defined by a particular choice of the \textbf{variational parameters} (HFPs)
	\[
	\Delta_{\alpha\beta}^{+-} = \langle 
	\hat c_\alpha^\dagger \hat c_\beta
	\rangle \, ,
	\qquad
	\Delta_{\alpha\beta}^{++} = \langle 
	\hat c_\alpha^\dagger \hat c_\beta^\dagger
	\rangle \, .
	\]
	Let $\mathbf{v}$ be the vector that collects all HFPs. We aim to find
	\[
	\ket{\Psi[\mathbf{v}]} \simeq \ket{\Psi_0} \, ,
	\]
	with $\mathbf{v}$ such that:
	
	\begin{columns}[t]
		\column{0.45\linewidth}
		\begin{tabbox}[frametitle=Variational picture]
			Energy is minimised,
			\[
			\fdv{E[\Phi]}{\ket{\Phi}} \bigg|_{\ket{\Phi} = \ket{\Psi[\mathbf{v}]}} = 0 \, .
			\]
		\end{tabbox}
		\column{0.45\linewidth}
		\begin{tabbox}[frametitle=Self-consistency picture]
			A self-consistency problem is satisfied,
			\[
			\mathrm{A}(\mathbf{v}) = \mathbf{v} \, ,
			\]
			being $\mathrm{A}$ a non-linear map \textbf{uniquely determined}.
		\end{tabbox}
	\end{columns}
\end{frame}

\begin{frame}
	\frametitle{The iterative algorithm (iHF)}
	\begin{columns}
		\column{0.45\linewidth}
		The HF solution is the \textbf{fixed point} of a non-linear map $\mathrm{A}$,
		\[
		\mathrm{A}(\mathbf{v}) = \mathbf{v} \, .
		\]
		\DownImplication
		\textbf{Strategy}: ``follow'' the map up to the fixed point,
		\[
		\mathbf{v}_{i+1} \equiv g \mathrm{A}(\mathbf{v}_i) + (1-g) \mathbf{v}_i \, .
		\]
		with $i>0$, $g \in [0,1]$.
		\DownImplication
		\column{0.45\linewidth}
		\begin{figure}
			\centering
			\includegraphics[width=\columnwidth]{/home/nepero27178/Thesis/EHM-HF/Project/src/labs/attractor/example-attractor-ink.pdf}
			\caption*{Example path for the AF phase.}
		\end{figure}
	\end{columns}
	\textbf{Convergence criterion}:
	\[
	\forall j \quad\colon\quad
	\left[ \mathbf{v}_{i+1} - \mathbf{v}_i \right]_j < \Delta v_j \, ,
	\]
	with $\Delta \mathbf{v}$ the vector of tolerances.
\end{frame}
