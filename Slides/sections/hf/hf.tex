\section{The Hartree-Fock method}

\begin{frame}
	\frametitle{The constrained minimisation framework}
	\begin{columns}
		\column{0.55\linewidth}
		We want to approximate the true ground state $\ket{\Psi_0}$ of the EHM, with energy $E_0$.\medskip
		
		\textbf{Variational principle}:
		\[
			\forall \ket{\Phi} \in \mathcal{S} \subset \mathcal{H} \quad\colon\quad
			E[\Phi] \ge E_0 \, .
		\]
		\DownImplication
		\begin{tabbox}[frametitle=Constrained minimisation procedure]
			\[
				\ket{\Psi} \quad\colon\quad
				\fdv{E[\Phi]}{\ket{\Phi}} \bigg|_{\ket{\Phi} = \ket{\Psi}} = 0 \, .
			\]
			The state $\ket{\Psi}$ is the best approximation to $\ket{\Psi_0}$ inside of $\mathcal{S}$.
		\end{tabbox}		
		\column{0.35\linewidth}
		\begin{figure}
			\centering
			\resizebox{\columnwidth}{!}{
			\input{../Thesis/chapters/hf-introduction/pictures/gaussian-subspace}
			}
			\caption*{Sketch of the Hilbert space $\mathcal{H}$ and a subset $\mathcal{S}$.}
		\end{figure}
	\end{columns}
\end{frame}

\begin{frame}
	\frametitle{The Hartree-Fock method in a nutshell}
	
	\begin{tabbox}[frametitle=The Hartree-Fock subspace of states] 
		$\mathcal{S}$ is the set of all \textbf{gaussian states},
		\[
		\ket{\Psi} = \bigotimes_\alpha \hat \gamma_\alpha^\dagger \ket{\Omega_\gamma} \, ,
		\qquad
		\acomm{\hat \gamma_\alpha}{\hat \gamma_\beta^\dagger} = \delta_{\alpha\beta} \, ,
		\]
		with $\hat \gamma \leftrightarrow \hat c$ through a unitary map.
	\end{tabbox}
	
	\uncover<2>{
	
	\medskip
	\DownImplication
	\medskip
	
	Each gaussian state can be identified \textbf{uniquely} by fixing
	\[
		\Delta_{\alpha\beta}^{+-} = \langle 
		\hat c_\alpha^\dagger \hat c_\beta
		\rangle \, ,
		\qquad
		\Delta_{\alpha\beta}^{++} = \langle 
		\hat c_\alpha^\dagger \hat c_\beta^\dagger
		\rangle \, .
	\]
	These are the \textbf{variational parameters} collected in the vector $\mathbf{v}$. We aim to find the optimal $\mathbf{v}$, such that
	\[
		\ket{\Psi[\mathbf{v}]} \simeq \ket{\Psi_0} \, .
	\]
	
	}

\end{frame}
