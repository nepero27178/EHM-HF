\section{The Hartree-Fock method}

\begin{frame}
	\frametitle{The constrained minimisation framework}
	\begin{columns}
		\column{0.55\linewidth}
		We want to approximate the true ground state $\ket{\Psi_0}$ of the EHM, with energy $E_0$.\medskip
		
		\textbf{Variational principle}:
		\[
			\forall \ket{\Phi} \in \mathcal{S} \subset \mathcal{H} \quad\colon\quad
			E[\Phi] \ge E_0 \, .
		\]
		\DownImplication
		\textbf{Constrained minimisation} procedure:
		\[
			\ket{\Psi} \quad\colon\quad
			\fdv{E[\Phi]}{\ket{\Phi}} \bigg|_{\ket{\Phi} = \ket{\Psi}} = 0 \, .
		\]
		The state $\ket{\Psi}$ is the best approximation to $\ket{\Psi_0}$ inside of $\mathcal{S}$.

	\end{columns}
\end{frame}

\begin{frame}
	\frametitle{The Hartree-Fock method in a nutshell}
	
	\textbf{Hartree-Fock method}: we identify $\mathcal{S}$ with the set of all the ground states of non-interacting hamiltonians,
	\[
		\mathcal{S} = \left\lbrace
			\ket{\Psi} \,\mid\, \text{$\ket{\Psi}$ is the ground-state of some $\hat H_\mathrm{HF}$}
		\right\rbrace \, ,
	\]
	where 
	\[
	\hat H_\mathrm{HF} \equiv
	\sum_{\alpha} E_\alpha \hat\gamma_\alpha^\dagger \hat\gamma_\alpha 
	\qq{with}
	\acomm{\hat \gamma_\alpha}{\hat \gamma_\beta^\dagger} = \delta_{\alpha\beta} \, ,
	\]
	with $\hat \gamma \leftrightarrow \hat c$ through a unitary map.\medskip
	\DownImplication
	\textbf{Wick's theorem}:
	\[
		\langle
		\hat c_\alpha^\dagger \hat c_\beta^\dagger \hat c_\gamma \hat c_\delta
		\rangle = \underbrace{
			\langle 
			\hat c_\alpha^\dagger \hat c_\delta
			\rangle \langle	
			\hat c_\beta^\dagger \hat c_\gamma 
			\rangle
		}_{\text{Hartree}}
		- 
		\underbrace{
			\langle 
			\hat c_\alpha^\dagger \hat c_\gamma
			\rangle \langle	
			\hat c_\beta^\dagger \hat c_\delta 
			\rangle 
		}_{\text{Fock}}
		+ 
		\underbrace{
			\langle 
			\hat c_\alpha^\dagger \hat c_\beta^\dagger
			\rangle \langle	
			\hat c_\gamma \hat c_\delta 
			\rangle 
		}_{\text{Cooper}}.
	\]
\end{frame}

\begin{frame}
	\frametitle{The self-consistency mapping}
	Target parametric hamiltonian:
	\[
		\hat H_\mathrm{HF}[\mathbf{v}] \equiv
		\sum_{\alpha} E_\alpha [\mathbf{v}] \hat\gamma_\alpha^\dagger [\mathbf{v}] \hat\gamma_\alpha [\mathbf{v}] 
		\qquad
		\text{with ground state $\ket{\Psi[\mathbf{v}]}$,}
	\]
	being $\mathbf{v}$ a vector of variational parameters. HF approximation:
	\[
		\ket{\Psi_0} \simeq \ket{\Psi[\mathbf{v}]} \, ,
	\]
	with $\mathbf{v}$ such that:
	
	\begin{columns}[t]
		\column{0.45\linewidth}
			\begin{center}
			\textbf{Variational picture}
			\end{center}
			Energy is minimised,
			\[
				\fdv{E[\Phi]}{\ket{\Phi}} \bigg|_{\ket{\Phi} = \ket{\Psi[\mathbf{v}]}} = 0 \, .
			\]
		\column{0.45\linewidth}
			\begin{center}
			\textbf{Self-consistency picture}
			\end{center}
			A self-consistency problem is satisfied,
			\[
				\mathrm{A}(\mathbf{v}) = \mathbf{v} \, ,
			\]
			being $\mathrm{A}$ a non-linear map \textbf{to be determined}.
	\end{columns}
	
\end{frame}

\begin{frame}
	\frametitle{The iterative algorithm (iHF)}
	\begin{columns}
		\column{0.45\linewidth}
	The HF solution is the \textbf{fixed point} of a non-linear map $\mathrm{A}$,
	\[
		\mathrm{A}(\mathbf{v}) = \mathbf{v} \, .
	\]
	\DownImplication
	\textbf{Strategy}: ``follow'' the map up to the fixed point,
	\[
		\mathbf{v}_{i+1} \equiv g \mathrm{A}(\mathbf{v}_i) + (1-g) \mathbf{v}_i \, .
	\]
	with $i>0$, $g \in [0,1]$.
