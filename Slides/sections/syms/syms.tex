\begin{frame}
	\frametitle{The symmetries of the EHM}
	\textbf{The model (EHM)}:
	\begin{equation*}
		\hat H_\mathrm{EHM} = 
		\underbrace{
			-t \sum_{\langle ij \rangle} \sum_\sigma \hat c_{i\sigma}^\dagger \hat c_{j\sigma}
		}_\text{Kinetic operator}
		\underbrace{
			+
			\vphantom{\sum_{\langle ij \rangle}}
			U \sum_i \hat n_{i\uparrow} \hat n_{i\downarrow}
		}_\text{Local repulsion} 
		\underbrace{
			-V \sum_{\langle ij \rangle} \hat n_i \hat n_j
		}_\text{Non-local attraction}
		\qquad
		\text{with $t, U, V  > 0$,}
	\end{equation*}
	and its symmetries:
	\begin{itemize}
		\item \textit{Translational invariance} in directions $x$ and $y$, $\mathrm{T}_1^{(x)} \otimes \mathrm{T}_1^{(y)}$.\\
		\textbf{If broken}: Non-uniform spin density.
		\item \textit{Rotational invariance} under $\pi/2$ rotations, $\mathrm{C}_4$.\\
		\textbf{If broken}: nematicity.
		\item \textit{Gauge invariance} (charge conservation), $\mathrm{U}(1)$.\\
		\textbf{If broken}: Superconductivity.
		\item \textit{Global spin rotational invariance}, $\mathrm{SU}(2)$.\\
		\textbf{If broken}: Magnetism.	
		\end{itemize}
\end{frame}
