\section{Conclusions and outlook}

\begin{frame}
	\frametitle{Our original results, summarised}
	
	\textbf{We wanted to} understand how the new interaction relates to Mott physics, antiferromagnetism and superconductivity.
	\vspace{1em}
	
	\textbf{We conclude that} at HF level, the non-local attraction:
	\begin{itemize}
		\item Acts as a source of Mott localisation;
		\item Enhances antiferromagnetism at half filling;
		\item Produces various ``flavours'' of superconductivity;
		\item Produces nematic superconductivity.
	\end{itemize}
	
\end{frame}

\begin{frame}
	\frametitle{Further developments}
	
	From here, we can follow various paths:
	\begin{enumerate}
		\item \textbf{Same framework, other phases}.\\
		Triplet superconductivity, complex breaking of translational invariance, nematic antiferromagnetism\dots
		\item \textbf{Expansion of the model}.\\
		Second nearest neighbours hopping, more refined potentials\dots
		\item \textbf{More advanced numerical techniques}.\\
		DMRG for chains and ladders; hybrid DMFT (local repulsion $U$) and HF (non-local attraction $-V$) to retrieve Mott physics induced by $U$\dots
	\end{enumerate}
\end{frame}
