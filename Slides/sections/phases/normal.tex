\section{The normal phase}

\begin{frame}
	\frametitle{The self-consistency problem for the normal phase}
	\textbf{Normal phase}: no broken symmetries. From the Fock channel,
	\[
	\tilde{t}_{ij\sigma} \equiv
	\begin{cases}
		\displaystyle t - V \langle \hat c_{j\sigma}^\dagger \hat c_{i\sigma} \rangle
		\qquad
		&\text{if $i,j$ are NNs,} \\
		0 &\text{otherwise.}
	\end{cases}	
	\]
	\DownImplication
	\textbf{Reciprocal space}: let $u_\mathbf{k} \equiv \langle \hat n_\mathbf{k} \rangle$ be the occupation and $u^{(\gamma)}$ its harmonics,
	\[
		\tilde{t}^{(s^*)} = t - \frac{u^{(s^*)} V}{2} \, .
	\]
\end{frame}
