\section{The normal phase}

\begin{frame}
	\frametitle{The variational problem for the normal phase}
	\textbf{Normal phase}: no broken symmetries. Just one variational parameter,
	\[
		\mathbf{v} = \left[ u^{(s^*)} \right] \, ,
	\]
	where $u^{(s^*)}$ is the $s^*$-wave component of $u_{\mathbf{k}\sigma} \equiv \ev{\hat c_{\mathbf{k}\sigma}^\dagger \hat c_{\mathbf{k}\sigma}}$.
	\DownImplication
	
	\begin{tabbox}[frametitle=Renormalisation of kinetics]
		Hopping amplitude is \textbf{isotropically renormalised},
		\[
			t \quad\to\quad \tilde{t}^{(s^*)} \equiv t - \frac{u^{(s^*)}V}{2} \, ,
		\]
		with the \textbf{theoretical constraint}:
		\[
		0 < \tilde{t}^{(s^*)} < t
		\quad\implies\quad
		\text{Possible Mott localisation.}
		\]
	\end{tabbox}
\end{frame}

\begin{frame}
	\frametitle{What we found: $V$-induced Mott physics}
	\begin{tabbox}[frametitle=The original results]
		\begin{itemize}
			\item Mott localisation induced by $-V$;
			\item Analytical phase boundary for the Mott transition in the $(\delta,V)$ plane.
		\end{itemize}
	\end{tabbox}
	
	\bigskip
	
	\textbf{Remarks}:
	\begin{itemize}
		\item Our solution is independent of $U$, which is generally the source of Mott localisation.
		\item At Hartree-Fock level, the Hubbard model ($V=0$) cannot describe Mott physics.
	\end{itemize}
\end{frame}
