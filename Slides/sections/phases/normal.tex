\section{The normal phase}

\begin{frame}
	\frametitle{The self-consistency problem for the normal phase}
	\textbf{Normal phase}: no broken symmetries. Just one variational parameter,
	\[
		\mathbf{v} = \left[ u^{(s^*)} \right] \, ,
	\]
	where $u^{(s^*)}$ is the $s^*$-wave component of $u_{\mathbf{k}\sigma} \equiv \ev{\hat c_{\mathbf{k}\sigma}^\dagger \hat c_{\mathbf{k}\sigma}}$.
	\DownImplication
	
	\begin{tabbox}[frametitle=Kinetics renormalisation]
		Hopping amplitude is \textbf{isotropically renormalised},
		\[
			t \quad\to\quad \tilde{t}^{(s^*)} \equiv t - \frac{u^{(s^*)}V}{2} \, .
		\]
		\DownImplication
		Renormalisation of electronic bands, $\epsilon_{\mathbf{k}\sigma} \to \tilde{\epsilon}_{\mathbf{k}\sigma}$.
	\end{tabbox}
\end{frame}

\begin{frame}	
	\frametitle{Mott localisation in the normal phase}
	\textbf{Theoretical constraint}:
	\[
	0 < \tilde{t}^{(s^*)} < t
	\quad\implies\quad
	\text{\textbf{Flattening} of bands = Mott localisation.}

\begin{frame}
	\frametitle{The HF solution in the normal phase}
	The \textbf{normal phase}, summarised:
	\begin{itemize}
		\item Preserves all symmetries of the EHM;
		\item The HF approximation is \textbf{independent of $U$};
		\item Presence of $V$-induced Mott localisation and \textbf{bands flattening};
		\item We identify a Mott transition boundary as we vary the doping level.
	\end{itemize}
\end{frame}
