\section{The normal phase}

\begin{frame}
	\frametitle{The self-consistency problem for the normal phase}
	\textbf{Normal phase}: no broken symmetries. From the Fock channel,
	\[
	\tilde{t}_{ij\sigma} \equiv
	\begin{cases}
		\displaystyle t - V \langle \hat c_{j\sigma}^\dagger \hat c_{i\sigma} \rangle
		\qquad
		&\text{if $i,j$ are NNs,} \\
		0 &\text{otherwise.}
	\end{cases}	
	\]
	\DownImplication
	\textbf{Self-consistency problem}: one HFP,
	\[
		u^{(s^*)} = \text{($s^*$-wave component of $u_\mathbf{k} \equiv \ev{\hat n_\mathbf{k}}$)} \, ,
	\]
	s.t. hopping amplitude is isotropically renormalised,
	\[
		t \quad\to\quad \tilde{t}^{(s^*)} \equiv t - \frac{u^{(s^*)}V}{2} \, .
	\]\medskip
	
	Therefore: renormalisation of electronic bands, $\epsilon_{\mathbf{k}\sigma} \to \tilde{\epsilon}_{\mathbf{k}\sigma}$.
\end{frame}

	\[
	0 < \tilde{t}^{(s^*)} < t
	\quad\implies\quad
	\text{\textbf{Flattening} of bands = Mott localisation.}

\begin{frame}
	\frametitle{The HF solution in the normal phase}
	The \textbf{normal phase}, summarised:
	\begin{itemize}
		\item Preserves all symmetries of the EHM;
		\item The HF approximation is \textbf{independent of $U$};
		\item Presence of $V$-induced Mott localisation and \textbf{bands flattening};
		\item We identify a Mott transition boundary as we vary the doping level.
	\end{itemize}
\end{frame}
