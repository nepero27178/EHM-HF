\section{The normal phase}

\begin{frame}
	\frametitle{The variational problem for the normal phase}
	\textbf{Normal phase}: no broken symmetries. Just one variational parameter,
	\[
		\mathbf{v} = \left[ u^{(s^*)} \right] \, \text{,}
	\]
	where $u^{(s^*)}$ is the $s^*$-wave component of $u_{\mathbf{k}\sigma} \equiv \ev{\hat c_{\mathbf{k}\sigma}^\dagger \hat c_{\mathbf{k}\sigma}}$.
	\DownImplication
	
	\begin{tabbox}[frametitle=Renormalisation of kinetics]
		The hopping amplitude $t$ is \textbf{isotropically renormalised},
		\[
			t \quad\to\quad \tilde{t}^{(s^*)} \equiv t - \frac{u^{(s^*)}V}{2} \, \text{.}
		\]
		We have proved the \textbf{constraint}:
		\[
		0 < \tilde{t}^{(s^*)} < t
		\quad\implies\quad
		\text{Possible Mott localisation.}
		\]
	\end{tabbox}
\end{frame}

\begin{frame}
	\frametitle{The original results}
	The role of $V/t$:
	\begin{itemize}
		\item \textbf{Renormalise hopping};
		\item \textbf{Induce complete Mott localisation}:
		\[
			\text{Effective kinetics energy} \ll \text{Interaction energy} \, .
		\]
	\end{itemize}
	The role of $U/t$:
	\begin{itemize}
		\item At HF level, $U$ does not produce Mott localisation.
	\end{itemize}
\end{frame}
