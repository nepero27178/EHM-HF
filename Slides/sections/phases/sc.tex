\section{The superconducting phase}

\begin{frame}
	\frametitle{The translationally invariant SC phase}
	
	Cooper pairing:
	\begin{itemize}
		\item Described by a SC gap $\tilde{\Delta}_{ij}$, roughly the ``binding energy'' of the pairs;
		\item The pair spin obeys sum rules:
		\[
			\frac 1 2 \otimes \frac 1 2 = 
				\only<1>{0}
				\only<2->{\mathbf{0}} 
			\oplus 1 = 
				\only<1>{\text{singlet}}
				\only<2->{\text{\bfseries singlet}} 
			\oplus \text{triplet.}
		\]
	\end{itemize}
	
	\uncover<2->{
		\DownImplication
	
		\begin{tabbox}[frametitle=The singlet channel]
			The gap function is inversion symmetric, i.e. its harmonics are
			\[
				\tilde{\Delta}^{(s)} \, , \quad \tilde{\Delta}^{(s^*)}\qq{and} \tilde{\Delta}^{(d)} \, .
			\]
		\end{tabbox}
	}
\end{frame}

\begin{frame}
	\frametitle{The variational problem for the singlet SC phase}
	
	We limit to the \textbf{singlet channel}. Let the quantities:
	\[
		u_{\mathbf{k}\sigma} \equiv \ev{\hat c_{\mathbf{k}\sigma}^\dagger \hat c_{\mathbf{k}\sigma}}
		\, , \quad
		w_\mathbf{k} \equiv \ev{\hat c_{\mathbf{k}\uparrow}^\dagger \hat c_{-\mathbf{k}\downarrow}^\dagger} \, .
	\]
	\textbf{Variational parameters}: symmetric harmonics,
	\[
		\mathbf{v} = \left( u^{(s^*)}, u^{(d)}, w^{(s)}, w^{(s^*)}, w^{(d)} \right)
	\]	
	
	\bigskip
	
	\begin{tabbox}[frametitle=The role of each parameter]
		\begin{itemize}
			\item $u^{(s^*)}$, $u^{(d)}$: renormalisation of kinetics;
			\item $w^{(s)}$, $w^{(s^*)}$ and $w^{(d)}$: components of the SC gap.
		\end{itemize}
	\end{tabbox}
	
\end{frame}

\begin{frame}
	\frametitle{The competition among gap components}
	
	Isotropic gap components:
	\[
	\begin{aligned}
		\tilde{\Delta}^{(s^*)} &= Vw^{(s^*)} 
		\qquad\text{(Mediated by the isotropic non-local attraction)}\\
		\tilde{\Delta}^{(s)} &= -Uw^{(s)} \qquad\text{(Mediated by the \textbf{local repulsion})}
	\end{aligned}
	\]
	Anisotropic gap component:
	\[
		\tilde{\Delta}^{(d)} = Vw^{(d)} 
		\qquad\text{(Mediated by the isotropic non-local attraction)}
	\]

\begin{frame}
	\frametitle{The nematic order parameter}
	
	In the SC phase, the renormalisation of $t$ gives two components,
	\[
	t \quad\to\quad
	\tilde{t}^{(s^*)} \equiv t - \frac{u^{(s^*)} V}{2}
	\,\qq{and}\,
	\tilde{t}^{(d)} \equiv - \frac{u^{(d)} V}{2} \, .
	\]
	\uncover<2->{\begin{columns}
		\column{0.4\linewidth}
		\begin{figure}
			\centering
			\resizebox{\columnwidth}{!}{\input{/home/nepero27178/Thesis/EHM-HF/Thesis/chapters/normal-phase/pictures/anisotropic-hopping}}
		\end{figure}
		\column{0.5\linewidth}
		\begin{tabbox}[frametitle=Anisotropic hopping]
			Direction-resolved effective hopping:
			\[\begin{aligned}
				\tilde{t}^{(x)} &\equiv \frac{\tilde{t}^{(s^*)} + \tilde{t}^{(d)}}{2} \, , \\
				\tilde{t}^{(y)} &\equiv \frac{\tilde{t}^{(s^*)} - \tilde{t}^{(d)}}{2} \, .
			\end{aligned}\]
			\DownImplication
			$\tilde{t}^{(d)} \neq 0 \implies$ \textbf{nematic SSB}.
		\end{tabbox}
	\end{columns}}
	
\end{frame}

\begin{frame}
	\frametitle{The original results}
	The role of $V/t$:
	\begin{itemize}
		\item \textbf{Renormalise hopping anisotropically};
		\item \textbf{Induce mixed-symmetry superconductivity};
		\item \textbf{Induce nematic symmetry breaking}.
	\end{itemize}
	The role of $U/t$:
	\begin{itemize}
		\item \textbf{Participate in superconductivity even if locally repulsive}.
	\end{itemize}
\end{frame}
