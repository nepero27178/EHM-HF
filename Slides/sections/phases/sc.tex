\section{The superconducting phase}

\begin{frame}
	\frametitle{Singlet channel Cooper pairing symmetries}
	
	SC phase: breaking of charge conservation $\mathrm{U}(1)$, insurgence of a SC gap.
	\DownImplication
	Anomalous densities can be non-zero,
	\[
		\Delta_{\alpha\beta}^{++} = \langle 
		\hat c_\alpha^\dagger \hat c_\beta^\dagger
		\rangle \neq 0 \, .
	\]
	\medskip
	\begin{tabbox}[frametitle=HFPs of the SC phase]
		Harmonics of the conventional density,
		\[
			u_\mathbf{k} \equiv \ev{\hat c_{\mathbf{k}\sigma}^\dagger \hat c_{\mathbf{k}\sigma}}
			\quad\to\quad
			u^{(\gamma)} \, ,
		\]
		Harmonics of the anomalous density,
		\[
			w_\mathbf{k} \equiv \ev{\hat c_{\mathbf{k}\uparrow}^\dagger \hat c_{-\mathbf{k}\downarrow}^\dagger}
			\quad\to\quad
			w^{(\gamma)} \, .
		\]
	\end{tabbox}
	
\end{frame}

\begin{frame}
	\frametitle{Symmetry mixing and insurgence of nematicity (1/2)}
	
	In the SC phase, the renormalisation of $t$ gives two components,
	\[
		t \quad\to\quad
		\tilde{t}^{(s^*)} \equiv t - \frac{u^{(s^*)} V}{2}
		\qq{and}
		\tilde{t}^{(d)} \equiv - \frac{u^{(d)} V}{2} \, .
	\]
	\begin{columns}
		\column{0.4\linewidth}
		\begin{figure}
			\centering
			\resizebox{\columnwidth}{!}{\input{/home/nepero27178/Thesis/EHM-HF/Thesis/chapters/normal-phase/pictures/anisotropic-hopping}}
		\end{figure}
		\column{0.5\linewidth}
		\begin{tabbox}[frametitle=Anisotropic hopping]
			Direction-resolved effective hopping:
			\[\begin{aligned}
				\tilde{t}^{(x)} &\equiv \frac{\tilde{t}^{(s^*)} + \tilde{t}^{(d)}}{2} \, , \\
				\tilde{t}^{(y)} &\equiv \frac{\tilde{t}^{(s^*)} - \tilde{t}^{(d)}}{2} \, .
			\end{aligned}\]
			\DownImplication
			$\tilde{t}^{(d)} \neq 0 \implies$ \textbf{nematic SSB}.
		\end{tabbox}
	\end{columns}
	
\end{frame}
