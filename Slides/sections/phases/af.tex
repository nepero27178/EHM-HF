\section{The antiferromagnetic phase}

\begin{frame}
	\frametitle{Commensurate antiferromagnetism in cuprates}
	
	\begin{columns}
		\column{0.3\linewidth}
		\begin{figure}
			\centering
			\input{\SectionFolder/commensurate-af.tex}
		\end{figure}
		\smallskip
		
		\column{0.6\linewidth}
		In cuprates, the AF phase is \textbf{commensurate}:
		\begin{itemize}
			\item $2$-sites periodic in each direction;
			\item Sublattices oppositely magnetised;
			\item Zero net magnetisation.
		\end{itemize}
		\DownImplication
		\textbf{Staggered magnetisation} $m$:
		\[
		2m = \abs{\ev{\hat n_{i\uparrow}} - \ev{\hat n_{i\downarrow}}} \, ,
		\]
		with $i$ a site.
		
		
	\end{columns}
	
	\uncover<2>{
		\begin{tabbox}[frametitle=Half-filling restriction]
			We limit to $\delta=0$ (half-filling) due to \textbf{stability considerations}.
		\end{tabbox}
	}
	
\end{frame}

\begin{frame}
	\frametitle{The variational problem for the AF phase}
	
	\textbf{Variational parameters} for the AF phase:
	\[
	\mathbf{v} = \begin{bmatrix}
		m \\ u^{(s^*)}
	\end{bmatrix} \, ,
	\]
	with $u^{(s^*)}$ defined as for the normal phase. \bigskip
	
	\begin{tabbox}[frametitle=Renormalisation of kinetics]
		Mott localisation effect \textbf{formally} identical to the normal phase:
		\[
		t \quad\to\quad \tilde{t}^{(s^*)} \equiv t - \frac{u^{(s^*)}V}{2} \, ,
		\]
		with the \textbf{theoretical constraint}: $0 < \tilde{t}^{(s^*)} < t$.
	\end{tabbox}
	
\end{frame}

\begin{frame}
	\frametitle{The cooperation on magnetisation}
	
	\begin{figure}
		\centering
		\input{\SectionFolder/magnetisation}
		\caption{Numerical results for $m$. The gray curves indicate isovalued lines.}
	\end{figure}
	
	\begin{columns}
		\column{0.45\linewidth}
		\uncover<3>{
			\begin{tabbox}[frametitle={$V \gg U,t$}]
				New ``unconventional'' source of magnetisation, induced by the ratio $V/t$.
			\end{tabbox}
		}
		
		\column{0.45\linewidth}
		\uncover<2>{
			\begin{tabbox}[frametitle={$U \gg V,t$}]
				``Conventional'' Hubbard magnetisation, induced by the ratio $U/t$.
			\end{tabbox}
		}
	\end{columns}
	
\end{frame}

\begin{frame}
	\frametitle{The pairing enhancement of magnetisation}	
	\begin{columns}
		\column{0.6\linewidth}
		\begin{figure}
			\centering
			\subfloat{
				\includegraphics[width=\columnwidth]{../Project/src/labs/antiferromagnet-iso-heatmaps/af-m-data-C128.pdf}
			}\\
			\subfloat{
				\includegraphics[width=\columnwidth]{../Project/src/labs/antiferromagnet-iso-heatmaps/af-tS-data-C128.pdf}
			}
			\caption{Numerical results for $m$ (top) and $\tilde{t}^{(s^*)}$ (bottom). The gray curves indicate isovalued lines.}
		\end{figure}
		\column{0.3\linewidth}
		At low temperature:
		\[
			m \simeq \frac{1}{2} + \mathcal{O} \left(r^2\right) \, ,
		\]
		where $r=\tilde{t}^{(s^*)}/U$.
		\uncover<2>{
			\medskip\DownImplication\medskip
			
			To enhance $m$, either:
			\begin{itemize}
				\item Increase $U$;
				\item \textbf{Decrease} $\tilde{t}^{(s^*)}$ .
			\end{itemize}
		}
		
	\end{columns}
	
\end{frame}

\begin{frame}
	\frametitle{Mott localisation in the AF phase}
	\begin{columns}
		\column{0.3\linewidth}
		
		Effective hopping is:
		\begin{itemize}
			\item reduced by $V$;
			\item \textbf{increased} by $U$.
		\end{itemize}
		
		\medskip
		\DownImplication
		\medskip
		
		The two interactions ``oppose'' each other.
		\column{0.6\linewidth}
		\begin{figure}
			\centering
			\includegraphics[width=\columnwidth]{../Project/src/labs/compare-tS-normal-af/tS(N)+tS(AF)-ink.pdf}
			\caption{Comparative plot of $\tilde{t}^{(s^*)}$ for normal and AF phase.}
		\end{figure}
	\end{columns}
\end{frame}

\begin{frame}
	\frametitle{The original results}
	The role of $V/t$:
	\begin{itemize}
		\item \textbf{Renormalise hopping};
		\item \textbf{Magnetise the lattice using Mott physics};
	\end{itemize}
	The role of $U/t$:
	\begin{itemize}
		\item Magnetise the lattice, conventional Hubbard physics.
	\end{itemize}
	
	\bigskip
	\DownImplication
	\bigskip
	
	\textbf{Cooperation} among interactions to magnetise the lattice.
\end{frame}
