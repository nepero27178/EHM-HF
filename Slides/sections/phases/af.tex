\begin{frame}
	\frametitle{Mott localisation in the AF phase}
	\begin{columns}
		\column{0.5\linewidth}
			\begin{figure}
				\centering
				\includegraphics[width=\columnwidth]{../Project/src/labs/compare-tS-normal-af/tS(N)+tS(AF)-ink.pdf}

		\column{0.4\linewidth}
			Setting $V>0$, we get the new HFP $u^{(s^*)}$ as for the normal phase,
			\[
			t \to \tilde{t}^{(s^*)} \equiv t - \frac{u^{(s^*)}V}{2} \, .
			\]\medskip
			
			\textbf{Theoretical constraint}:
			\[
			0 < \tilde{t}^{(s^*)} < t
			\]
			\DownImplication
	\end{columns}
	\begin{tabbox}[frametitle=Mott localisation in the AF phase]
		The local repulsion $U$ \textbf{opposes bands flattening} $\implies$ counter-intuitive!
	\end{tabbox}
\end{frame}

\begin{frame}
	\frametitle{The Mott enhancement of magnetisation}
	Magnetisation is \textbf{enhanced} by Mott localisation:
	\[
		0 < \tilde{t}^{(s^*)} < t
		\quad\implies\quad
		U/\tilde{t}^{(s^*)} > U/t \, .
	\]

\end{frame}

\begin{frame}
	\frametitle{The HF solution in the commensurate AF phase}
	The \textbf{AF phase}, summarised:
	\begin{itemize}
		\item Breaks spin rotation and translational invariance;
		\item The HF approximation is only stable at half-filling;
		\item Presence of $V$-induced Mott localisation and \textbf{bands flattening}, enhanced by $V$ and opposed by $U$;
		\item Two ways of magnetising the lattice: increasing $U$ (``conventional'') or using Mott effects (``unconventional'');
		\item (Not shown in slides) always energetically favoured compared to the normal phase.
	\end{itemize}
\end{frame}
