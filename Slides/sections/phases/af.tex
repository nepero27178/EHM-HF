\begin{frame}

			\bigskip

\begin{frame}
	\frametitle{The variational problem for the AF phase}
	
	Restrictions: commensurate AF phase, at half-filling.\medskip
	
	\textbf{Variational parameters} for the EHM:
	\[
		\mathbf{v} = \begin{bmatrix}
			m \\ u^{(s^*)}
		\end{bmatrix} \, ,
	\]
	with $u^{(s^*)}$ defined as for the normal phase. \medskip
	
	\begin{tabbox}[frametitle=Renormalisation of kinetics]
		Identical Mott localisation effect:
		\[
		t \quad\to\quad \tilde{t}^{(s^*)} \equiv t - \frac{u^{(s^*)}V}{2} \, ,
		\]
		with the \textbf{theoretical constraint}: $0 < \tilde{t}^{(s^*)} < t$.
	\end{tabbox}
	
\end{frame}

\begin{frame}
	\frametitle{What we found: cooperative magnetisation}
	\begin{tabbox}[frametitle=The original results]
		\begin{itemize}
			\item The non-local attraction $-V$ induces Mott localisation by reducing the hopping amplitude;
			\item Mott localisation \textbf{enhances magnetisation},
			\[
			0 < \tilde{t}^{(s^*)} < t
			\quad\implies\quad
			U/\tilde{t}^{(s^*)} > U/t \, .
			\]
			In the conventional HM, the ratio $U/t$ governs $m$.
		\end{itemize}
	\end{tabbox}
	
	\bigskip
	
	\textbf{Remarks}:
	\begin{itemize}
		\item $U$ opposes Mott localisation: Hartree-Fock artifact?
		\item $U$ and $V$ cooperate in magnetising the lattice: is there an ''optimal ratio''?
	\end{itemize}
\end{frame}
